{{דף של זהר|זהר חלק ב|ב א|ב א|ב ב|}}

וְאֵלֶּה שְׁמוֹת בְּנֵי יִשְׂרָאֵל הַבָּאִים מִצְרָיְמָה אֵת יַעֲקֹב אִישׁ וּבֵיתוֹ בָּאוּ, (דניאל י"ב) וְהַמַּשְׂכִּילִים יַזְהִירוּ כְּזֹהַר הָרָקִיעַ וּמַצְדִּיקֵי הָרַבִּים כַּכּוֹכָבִים לְעוֹלָם וָעֶד והַמַּשְׂכִּילִים: אִלֵּין אִינּוּן דְּמִסְתַּכְּלֵּי בְּרָזָא דְּחָכְמְתָא. (נ"א יזהירו דנצצין ברזא דחכמתא) יַזְהִירוּ: נַהֲרִין, וְנֹצְצִין בְּזִיוָא דְּחָכְמְתָא עִלָּאָה. כְּזֹהַר: נְהִירוּ וּנְצִיצוּ דְּנַהֲרָא דְּנָפִיק מֵעֵדֶן. וְדָא אִיהוּ רָזָא סְתִימָא, דְּאִקְרֵי (בראשית ח"ב ל"ד א') רָקִיעַ. בֵּיהּ קַיָימִין כּוֹכָבַיָא וּמַזָּלֵי שִׁמְשָׁא וְסִיהֲרָא, וְכָל אִינּוּן בּוּצִינִין דִּנְהוֹרָא. 

זֹהַר דְהַאי רָקִיעַ נָהִיר בִּנְהִירוּ עַל גִּנְתָּא. וְאִילָנָא דְּחַיֵּי קַיָּים בִּמְצִיעוּת גִּנְתָּא. דְּעַנְפּוֹי חַפְיָין (ד"א עליה) (נ"א על) כָּל אִינּוּן דִּיוּקְנִין וְאִילָנִין וּבוּסְמִין דִּבְגִּנְתָּא, בְּמָאנִין דְּכַשְׁרָן. (ס"א ומטללין) וּתְטַלְלִין תְּחוֹתֵיהּ כָּל חֵיוַת בָּרָא. וְכֹל צִפֳּרֵי שְׁמַיָא יְדוּרוּן תְּחוֹת אִינּוּן עַנְפִין. 

זֹהַר אִיבָּא דְּאִילָנָא, יָהִיב חַיִּין לְכֹלָּא. קִיוּמֵיהּ לְעָלַם וּלְעָלְמֵי עָלְמִין סִטְרָא אַחֲרָא לָא שַׁרְיָא בֵּיהּ, אֶלָּא סִטְרָא דִּקְדוּשָּׁה. זַכָּאָה חוּלָקֵיהוֹן אִינּוּן דְּטַעֲמִין מִנֵיהּ, אִינּוּן קַיְימִין לְעָלַם וּלְעָלְמֵי עָלְמִין. אִלֵּין אִקְרוּן מַשְׂכִּילִים, וְזַכָּאָן חַיִּין בְּהַאי עָלְמָא, וְחַיִּין בְּעָלְמָא דְּאָתֵי. 

זֹהַר אִילָנָא דָּא, זַקְפָא לְעֵילָּא לְעֵילָּא. חֲמֵשׁ מְאָה פַּרְסֵי הִלּוּכֵיהּ, שִׁתִּין רִבּוֹא אִיהוּ, בִּפְשִׁיטוּתֵיהּ. בְּהַאי אִילָנָא, קַיְּימָא חַד זֹהֲרָא, (זהר) כָּל גַּוְונִין קַיָּימִין בֵּיהּ אִינּוּן גַּוְונִין סָלְקִין וְנַחְתִּין, לָא מִתְיַשְׁבֵי בְּדוּכְתָּא אַחֲרָא בַּר בְּהַהוּא אִילָנָא. 

כַּד נַפְקֵי מִנֵּיהּ לְאִתְחֲזָאָה בְּגוֹ זֹהַר דְּלָא נַהֲרָא, מִתְיַשְּׁבָן וְלָא מִתְיַשְּׁבָן, קַיְימָן וְלָא קַיְימָן, בְּגִין דְּלָא מִתְיַשְּׁבָן בְּאֲתָר אַחֲרָא. מֵאִילָנָא דָּא נַפְקֵי תְּרֵיסַר שְׁבָטִין. דְּמִתְּחַמָּן בֵּיהּ, וְאִינּוּן נַחְתּוּ בְּהַאי זֹהַר דְּלָא נַהֲרָא, לְגוֹ גָּלוּתָא דְּמִצְרַיִם, בְּכַמָּה מַשִּׁירְיָין עִלָּאִין, הֲדָא הוּא דִכְתִיב וְאֵלֶּה שְׁמוֹת בְּנֵי יִשְׂרָאֵל וְגוֹ'.
